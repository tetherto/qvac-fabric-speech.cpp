cmake_minimum_required(VERSION 3.20)
project(qvac-eu-bench LANGUAGES C CXX OBJC OBJCXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    message(FATAL_ERROR "Configure with -DCMAKE_SYSTEM_NAME=iOS for a physical iPhone")
endif()
set(QVAC_GGML_SOURCE_DIR "" CACHE PATH "Current qvac-ext-ggml speech checkout")
if(NOT EXISTS "${QVAC_GGML_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Set QVAC_GGML_SOURCE_DIR to a current qvac-ext-ggml speech checkout")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(PARAKEET_COREML ON CACHE BOOL "" FORCE)
set(PARAKEET_INSTALL OFF CACHE BOOL "" FORCE)
set(PARAKEET_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PARAKEET_BUILD_EXECUTABLES OFF CACHE BOOL "" FORCE)
set(PARAKEET_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${QVAC_GGML_SOURCE_DIR}" ggml)
get_filename_component(PARAKEET_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${PARAKEET_ROOT}" parakeet)
# The explicit dependency must outrank an ignored, stale local ggml checkout.
target_include_directories(qvac-parakeet BEFORE PRIVATE "${QVAC_GGML_SOURCE_DIR}/include")
target_include_directories(parakeet.coreml BEFORE PRIVATE "${QVAC_GGML_SOURCE_DIR}/include")
add_executable(qvac-eu-bench MACOSX_BUNDLE main.mm
    "${PARAKEET_ROOT}/test/test_transcribepsy_eu_coreml.cpp")
target_compile_features(qvac-eu-bench PRIVATE cxx_std_17)
target_compile_options(qvac-eu-bench PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
target_compile_definitions(qvac-eu-bench PRIVATE PARAKEET_EU_COREML_TEST_LIBRARY)
target_link_libraries(qvac-eu-bench PRIVATE qvac-parakeet ggml
    "-framework UIKit" "-framework Foundation" "-framework CoreML")
